# 三、分销模块（核心）

---

## 📄 services/distributionService.js

```javascript
const db = require("../config/db");

// 绑定邀请关系（只允许一次）
async function bindRelation(inviterId, inviteeId) {
  if (inviterId == inviteeId) return;

  // 是否已绑定
  const [rows] = await db.execute(
    "SELECT * FROM DistributionRelation WHERE inviteeId=?",
    [inviteeId]
  );

  if (rows.length > 0) return;

  // 防循环（简单版）
  const [check] = await db.execute(
    "SELECT * FROM DistributionRelation WHERE inviterId=? AND inviteeId=?",
    [inviteeId, inviterId]
  );
  if (check.length > 0) return;

  await db.execute(
    "INSERT INTO DistributionRelation (inviterId, inviteeId, level) VALUES (?, ?, 1)",
    [inviterId, inviteeId]
  );
}

// 获取上级
async function getInviter(userId) {
  const [rows] = await db.execute(
    "SELECT inviterId FROM DistributionRelation WHERE inviteeId=?",
    [userId]
  );
  return rows.length ? rows[0].inviterId : null;
}

module.exports = {
  bindRelation,
  getInviter
};
```

# 四、分佣模块（赚钱核心）

---

## 📄 services/commissionService.js

```javascript
const db = require("../config/db");
const { getInviter } = require("./distributionService");

// 创建佣金记录
async function createCommission(userId, orderId, amount, level) {
  await db.execute(
    `INSERT INTO CommissionRecord 
    (userId, orderId, amount, level, status) 
    VALUES (?, ?, ?, ?, 'pending')`,
    [userId, orderId, amount, level]
  );
}

// 分佣逻辑（核心）
async function handleCommission(order) {
  const inviter = await getInviter(order.userId);

  if (inviter) {
    const amount1 = order.amount * 0.1;

    await createCommission(inviter, order.id, amount1, 1);

    const secondInviter = await getInviter(inviter);

    if (secondInviter) {
      const amount2 = order.amount * 0.03;

      await createCommission(secondInviter, order.id, amount2, 2);
    }
  }
}

module.exports = {
  handleCommission
};
```

# 七、支付回调（最关键）

---

## 📄 routes/payment.js

```javascript
const express = require("express");
const router = express.Router();
const db = require("../config/db");
const { handleCommission } = require("../services/commissionService");

// 支付回调
router.post("/callback", async (req, res) => {
  const { orderId, payStatus } = req.body;

  const conn = await db.getConnection();

  try {
    await conn.beginTransaction();

    // 查订单
    const [orders] = await conn.execute(
      "SELECT * FROM `Order` WHERE id=? FOR UPDATE",
      [orderId]
    );

    if (!orders.length) throw new Error("订单不存在");

    const order = orders[0];

    // 幂等：已支付直接返回
    if (order.status === "paid") {
      await conn.commit();
      return res.json({ success: true });
    }

    // 更新订单状态
    await conn.execute(
      "UPDATE `Order` SET status='paid', payTime=NOW() WHERE id=?",
      [orderId]
    );

    // 分佣
    await handleCommission(order);

    await conn.commit();

    res.json({ success: true });
  } catch (err) {
    await conn.rollback();
    res.status(500).json({ error: err.message });
  } finally {
    conn.release();
  }
});

module.exports = router;
```
