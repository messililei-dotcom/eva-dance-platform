## ✅ 完整代码

```vue
<template>
  <div class="distribution">
    <!-- 顶部数据卡片 -->
    <div class="card">
      <div class="item">
        <div class="value">¥{{ stats.totalIncome }}</div>
        <div class="label">累计收益</div>
      </div>
      <div class="item">
        <div class="value">¥{{ stats.todayIncome }}</div>
        <div class="label">今日收益</div>
      </div>
      <div class="item">
        <div class="value">{{ stats.inviteCount }}</div>
        <div class="label">邀请人数</div>
      </div>
    </div>

    <!-- 分享按钮 -->
    <div class="share-box">
      <button @click="handleShare">立即推广</button>
    </div>

    <!-- 分佣记录 -->
    <div class="list">
      <h3>分佣记录</h3>
      <div v-for="item in commissionList" :key="item.id" class="record">
        <div>订单ID: {{ item.orderId }}</div>
        <div>佣金: ¥{{ item.amount }}</div>
        <div>等级: {{ item.level }}级</div>
        <div>状态: {{ item.status }}</div>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref, onMounted } from "vue";
import axios from "axios";

const stats = ref({
  totalIncome: 0,
  todayIncome: 0,
  inviteCount: 0
});

const commissionList = ref([]);

const userId = 1001; // 实际应从登录态获取

// 获取统计数据
const getStats = async () => {
  const res = await axios.get(`/api/commission/stats?userId=${userId}`);
  stats.value = res.data;
};

// 获取分佣记录
const getCommissionList = async () => {
  const res = await axios.get(`/api/commission/list?userId=${userId}`);
  commissionList.value = res.data;
};

// 分享
const handleShare = async () => {
  const res = await axios.get(`/api/share/link?userId=${userId}`);
  const link = res.data.url;

  // H5方式
  navigator.clipboard.writeText(link);
  alert("推广链接已复制！");
};

onMounted(() => {
  getStats();
  getCommissionList();
});
</script>
```

# 二、分享组件（核心转化入口）

---

## 📦 ShareComponent.vue

```vue
<template>
  <div>
    <button @click="generateLink">生成推广链接</button>

    <div v-if="link" class="box">
      <input :value="link" readonly />
      <button @click="copy">复制</button>
    </div>
  </div>
</template>

<script setup>
import { ref } from "vue";
import axios from "axios";

const link = ref("");
const userId = 1001;

const generateLink = async () => {
  const res = await axios.get(`/api/share/link?userId=${userId}`);
  link.value = res.data.url;
};

const copy = () => {
  navigator.clipboard.writeText(link.value);
  alert("复制成功");
};
</script>

<style scoped>
.box {
  margin-top: 10px;
}
input {
  width: 100%;
  padding: 8px;
}
</style>
```

## 📄 pages/distribution/index.js

```javascript
Page({
  data: {
    stats: {},
    list: []
  },

  onLoad() {
    this.getStats();
    this.getList();
  },

  getStats() {
    wx.request({
      url: '/api/commission/stats',
      success: (res) => {
        this.setData({ stats: res.data });
      }
    });
  },

  getList() {
    wx.request({
      url: '/api/commission/list',
      success: (res) => {
        this.setData({ list: res.data });
      }
    });
  },

  onShareAppMessage() {
    return {
      title: "舞蹈课程推荐",
      path: `/pages/home/index?inviterId=1001`
    };
  }
});
```
